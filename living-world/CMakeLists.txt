cmake_minimum_required(VERSION 3.25)
project(FinalverseLivingWorldViewer VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
set(STORM_SOURCE_ROOT "" CACHE PATH "Accepted read-only Storm source checkout")
set(STORM_OGRE_BUILD_DIR "" CACHE PATH "Pinned prebuilt OGRE archives")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/storm-adapter.lock.json" lock)
string(JSON files LENGTH "${lock}" files)
math(EXPR last "${files}-1")
foreach(i RANGE ${last})
 string(JSON path GET "${lock}" files ${i} path)
 string(JSON expected GET "${lock}" files ${i} sha256)
 file(SHA256 "${STORM_SOURCE_ROOT}/${path}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Storm adapter source mismatch: ${path}; use the recorded frozen source candidate")
 endif()
endforeach()
string(JSON archives LENGTH "${lock}" ogre_archives)
math(EXPR archive_last "${archives}-1")
foreach(i RANGE ${archive_last})
 string(JSON path GET "${lock}" ogre_archives ${i} path)
 string(JSON expected GET "${lock}" ogre_archives ${i} sha256)
 file(SHA256 "${STORM_OGRE_BUILD_DIR}/${path}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "OGRE archive differs from the tested adapter lock: ${path}")
 endif()
endforeach()
add_subdirectory("${STORM_SOURCE_ROOT}/world-definition" storm-adapter)
find_package(SDL3 CONFIG REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(finalverse-living-render render.cpp shadows.cpp)
target_include_directories(finalverse-living-render SYSTEM PRIVATE "${STORM_SOURCE_ROOT}/third_party/rapidjson/include")
target_link_libraries(finalverse-living-render PRIVATE storm_fwd_scene storm_fwd_render SDL3::SDL3 OpenSSL::Crypto)
target_compile_options(finalverse-living-render PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-deprecated-declarations)
install(TARGETS finalverse-living-render)

if(BUILD_TESTING)
 add_executable(living-bridge-tests bridge_tests.cpp)
 target_include_directories(living-bridge-tests SYSTEM PRIVATE "${STORM_SOURCE_ROOT}/third_party/rapidjson/include")
 target_compile_options(living-bridge-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-deprecated-declarations)
 add_test(NAME living_bridge_contracts COMMAND living-bridge-tests)
endif()

option(LIVING_SANITIZERS "Instrument local bridge admission/control code with ASan/UBSan" OFF)
if(LIVING_SANITIZERS)
 foreach(target finalverse-living-render living-bridge-tests)
  if(TARGET ${target})
   target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
 endforeach()
endif()

target_include_directories(finalverse-living-render SYSTEM PRIVATE
 "${STORM_SOURCE_ROOT}/third_party/ogre-next/OgreMain/include"
 "${STORM_OGRE_BUILD_DIR}/third_party/ogre-next/include")

option(LIVING_FUZZ "Build the bounded scene/control parser libFuzzer target (Clang)" OFF)
if(LIVING_FUZZ)
 add_executable(living-bridge-fuzz fuzz_bridge.cpp)
 target_include_directories(living-bridge-fuzz SYSTEM PRIVATE "${STORM_SOURCE_ROOT}/third_party/rapidjson/include")
 # LLVM22 rejects an unused legacy RapidJSON GenericStringRef assignment
 # at template definition time. Defer only unused template bodies for this
 # fuzz translation unit; do not patch or replace the pinned vendor parser.
 target_compile_options(living-bridge-fuzz PRIVATE -fdelayed-template-parsing -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
 target_link_options(living-bridge-fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
