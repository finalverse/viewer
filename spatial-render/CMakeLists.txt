cmake_minimum_required(VERSION 3.25)
# FINALVERSE Viewer spatial renderer (Ω-XVIII): Storm scene contexts for the Viewer.
project(FinalverseSpatialRender VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
set(STORM_SOURCE_ROOT "" CACHE PATH "Storm checkout matching storm-adapter.lock.json")
set(STORM_OGRE_BUILD_DIR "" CACHE PATH "Pinned prebuilt OGRE archives")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/storm-adapter.lock.json" lock)
string(JSON files LENGTH "${lock}" files)
math(EXPR last "${files}-1")
foreach(i RANGE ${last})
 string(JSON path GET "${lock}" files ${i} path)
 string(JSON expected GET "${lock}" files ${i} sha256)
 file(SHA256 "${STORM_SOURCE_ROOT}/${path}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Storm source mismatch: ${path}; build against the locked candidate")
 endif()
endforeach()
string(JSON archives LENGTH "${lock}" ogre_archives)
math(EXPR archive_last "${archives}-1")
foreach(i RANGE ${archive_last})
 string(JSON path GET "${lock}" ogre_archives ${i} path)
 string(JSON expected GET "${lock}" ogre_archives ${i} sha256)
 file(SHA256 "${STORM_OGRE_BUILD_DIR}/${path}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "OGRE archive differs from the lock: ${path}")
 endif()
endforeach()
add_subdirectory("${STORM_SOURCE_ROOT}/world-definition" storm-adapter)
find_package(SDL3 CONFIG REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(finalverse-spatial-render spatial_render.cpp shadows.cpp)
target_include_directories(finalverse-spatial-render SYSTEM PRIVATE "${STORM_SOURCE_ROOT}/third_party/rapidjson/include"
 "${STORM_SOURCE_ROOT}/third_party/ogre-next/OgreMain/include" "${STORM_OGRE_BUILD_DIR}/third_party/ogre-next/include")
target_link_libraries(finalverse-spatial-render PRIVATE storm_fwd_scene storm_fwd_render SDL3::SDL3 OpenSSL::Crypto)
target_compile_options(finalverse-spatial-render PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-deprecated-declarations)
install(TARGETS finalverse-spatial-render)
# Ω-XIX native frame transport conformance (no GPU).
add_executable(frame_transport_test test_frame_transport.cpp)
target_compile_options(frame_transport_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME frame_transport COMMAND frame_transport_test)
add_executable(fvft-probe fvft_probe.cpp)
target_compile_options(fvft-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
# Ω-XX visual quality gates: read-only capture of the published frame.
add_executable(fvft-grab fvft_grab.cpp)
target_compile_options(fvft-grab PRIVATE -Wall -Wextra -Wpedantic -Werror)
